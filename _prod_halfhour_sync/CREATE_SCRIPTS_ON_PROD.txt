# Paste/run this on PROD in PowerShell.
# Purpose: RDP often blocks copying .cmd/.ps1 — recreate scripts locally.
# Also ensure push_service.py / push_report.py were copied (those .py usually paste OK).

$ErrorActionPreference = "Stop"
$ScriptsDir = "D:\deploy\testai_community_prod\backend\scripts"
if (-not (Test-Path $ScriptsDir)) { throw "Missing $ScriptsDir" }

function Write-Utf8NoBom([string]$Path, [string]$Content) {
    $utf8 = New-Object System.Text.UTF8Encoding $false
    [System.IO.File]::WriteAllText($Path, $Content, $utf8)
    Write-Host ("Wrote " + $Path)
}

Write-Utf8NoBom (Join-Path $ScriptsDir "install_halfhour_push_test.cmd") @'
@echo off
REM Install half-hour Daily/Weekly ALTERNATING test schedule (prod).
setlocal
cd /d "%~dp0"
echo Installing half-hour push TEST schedule ...
powershell -NoProfile -ExecutionPolicy Bypass -File "%~dp0install_halfhour_push_test.ps1"
if errorlevel 1 (
  echo FAIL: install_halfhour_push_test.ps1
  exit /b 1
)
echo.
echo REMEMBER: set DINGTALK_*_IDEMPOTENCY_ENABLED=false in prod .env
exit /b 0
'@

Write-Utf8NoBom (Join-Path $ScriptsDir "restore_normal_push_schedule.cmd") @'
@echo off
REM Restore normal push schedule after half-hour test.
setlocal
cd /d "%~dp0"
echo Restoring normal push schedule ...
powershell -NoProfile -ExecutionPolicy Bypass -File "%~dp0restore_normal_push_schedule.ps1"
if errorlevel 1 (
  echo FAIL: restore_normal_push_schedule.ps1
  exit /b 1
)
echo.
echo REMEMBER: set DINGTALK_*_IDEMPOTENCY_ENABLED=true in prod .env
exit /b 0
'@

Write-Utf8NoBom (Join-Path $ScriptsDir "wecom_push_halfhour_test.ps1") @'
$ErrorActionPreference = "Continue"
$ScriptsDir = $PSScriptRoot
$now = Get-Date
$minute = $now.Minute
$hour = $now.Hour
if ($hour -lt 10 -or $hour -gt 21 -or ($hour -eq 21 -and $minute -ge 30)) {
    Write-Host ("SKIP outside 10:00-21:00 window now={0:HH:mm}" -f $now)
    exit 0
}
$env:TM_PUSH_FORCE = "1"
if ($minute -ge 15 -and $minute -lt 45) {
    $kind = "weekly"
    $target = Join-Path $ScriptsDir "wecom_push_weekly.ps1"
} else {
    $kind = "daily"
    $target = Join-Path $ScriptsDir "wecom_push_daily.ps1"
}
Write-Host ("{0:yyyy-MM-dd HH:mm:ss} halfhour-test kind={1} force=1 script={2}" -f $now, $kind, $target)
& $target
exit $LASTEXITCODE
'@

Write-Utf8NoBom (Join-Path $ScriptsDir "install_halfhour_push_test.ps1") @'
$ErrorActionPreference = "Stop"
$ScriptsDir = $PSScriptRoot
$TestPs1 = Join-Path $ScriptsDir "wecom_push_halfhour_test.ps1"
if (-not (Test-Path $TestPs1)) { throw ("Missing " + $TestPs1) }
$DailyPs1 = Join-Path $ScriptsDir "wecom_push_daily.ps1"
$WeeklyPs1 = Join-Path $ScriptsDir "wecom_push_weekly.ps1"
if (-not (Test-Path $DailyPs1)) { throw ("Missing " + $DailyPs1) }
if (-not (Test-Path $WeeklyPs1)) { throw ("Missing " + $WeeklyPs1) }
$UserId = if ($env:USERDOMAIN) { ($env:USERDOMAIN + "\" + $env:USERNAME) } else { $env:USERNAME }
$PsExe = Join-Path $env:SystemRoot "System32\WindowsPowerShell\v1.0\powershell.exe"
$TestTaskName = "TestAI-Push-HalfHour-Test"
$DailyTaskName = "TestAI-WeCom-Daily"
$WeeklyTaskName = "TestAI-WeCom-Weekly"
Write-Host ("Test script : " + $TestPs1)
Write-Host ("User        : " + $UserId)
foreach ($tn in @($DailyTaskName, $WeeklyTaskName)) {
    $t = Get-ScheduledTask -TaskName $tn -ErrorAction SilentlyContinue
    if ($t) {
        Disable-ScheduledTask -TaskName $tn | Out-Null
        Write-Host ("Disabled: " + $tn)
    } else {
        Write-Host ("WARN missing task (skip disable): " + $tn)
    }
}
$arg = '-NoProfile -ExecutionPolicy Bypass -WindowStyle Hidden -File "' + $TestPs1 + '"'
$action = New-ScheduledTaskAction -Execute $PsExe -Argument $arg
$triggers = @()
for ($h = 10; $h -le 21; $h++) {
    $triggers += (New-ScheduledTaskTrigger -Daily -At ("{0:D2}:00" -f $h))
    if ($h -lt 21) {
        $triggers += (New-ScheduledTaskTrigger -Daily -At ("{0:D2}:30" -f $h))
    }
}
$settings = New-ScheduledTaskSettingsSet -AllowStartIfOnBatteries -DontStopIfGoingOnBatteries -StartWhenAvailable -WakeToRun -ExecutionTimeLimit (New-TimeSpan -Hours 1) -MultipleInstances IgnoreNew -Hidden
$principal = New-ScheduledTaskPrincipal -UserId $UserId -LogonType Interactive -RunLevel Limited
Get-ScheduledTask -TaskName $TestTaskName -ErrorAction SilentlyContinue | Unregister-ScheduledTask -Confirm:$false -ErrorAction SilentlyContinue
Register-ScheduledTask -TaskName $TestTaskName -Action $action -Trigger $triggers -Settings $settings -Principal $principal -Force | Out-Null
Write-Host ("Registered: " + $TestTaskName + " triggers=" + $triggers.Count)
Write-Host "REQUIRED: DINGTALK_PUSH_IDEMPOTENCY_ENABLED=false and DINGTALK_WEEKLY_IDEMPOTENCY_ENABLED=false"
'@

Write-Utf8NoBom (Join-Path $ScriptsDir "restore_normal_push_schedule.ps1") @'
$ErrorActionPreference = "Stop"
$ScriptsDir = $PSScriptRoot
$TestTaskName = "TestAI-Push-HalfHour-Test"
Get-ScheduledTask -TaskName $TestTaskName -ErrorAction SilentlyContinue | Unregister-ScheduledTask -Confirm:$false -ErrorAction SilentlyContinue
Write-Host ("Removed (if existed): " + $TestTaskName)
$Install = Join-Path $ScriptsDir "install_wecom_scheduled_tasks.ps1"
if (-not (Test-Path $Install)) { throw ("Missing " + $Install) }
& $Install
Write-Host "REQUIRED: set both DINGTALK_*_IDEMPOTENCY_ENABLED=true"
'@

# Patch daily/weekly FORCE preserve (ASCII-safe replace)
$daily = Join-Path $ScriptsDir "wecom_push_daily.ps1"
$weekly = Join-Path $ScriptsDir "wecom_push_weekly.ps1"
if (Test-Path $daily) {
    $t = Get-Content -LiteralPath $daily -Raw
    if ($t -notmatch 'IsNullOrWhiteSpace\(\$env:TM_PUSH_FORCE\)') {
        $t = $t -replace '\$env:TM_PUSH_FORCE = "0"', @'
if ([string]::IsNullOrWhiteSpace($env:TM_PUSH_FORCE)) {
    $env:TM_PUSH_FORCE = "0"
}
Write-Log ("TM_PUSH_FORCE=" + $env:TM_PUSH_FORCE)
'@
        Write-Utf8NoBom $daily $t
    } else {
        Write-Host "daily already patched"
    }
} else {
    Write-Host "WARN missing wecom_push_daily.ps1"
}
if (Test-Path $weekly) {
    $t = Get-Content -LiteralPath $weekly -Raw
    if ($t -notmatch 'IsNullOrWhiteSpace\(\$env:TM_PUSH_FORCE\)') {
        $t = $t -replace '\$env:TM_PUSH_FORCE = "0"', @'
if ([string]::IsNullOrWhiteSpace($env:TM_PUSH_FORCE)) {
    $env:TM_PUSH_FORCE = "0"
}
Write-Log ("TM_PUSH_FORCE=" + $env:TM_PUSH_FORCE)
'@
        Write-Utf8NoBom $weekly $t
    } else {
        Write-Host "weekly already patched"
    }
} else {
    Write-Host "WARN missing wecom_push_weekly.ps1"
}

Write-Host ""
Write-Host "DONE. Next:"
Write-Host "  1) ensure push_service.py + push_report.py copied"
Write-Host "  2) .env idempotency false"
Write-Host "  3) powershell -NoProfile -ExecutionPolicy Bypass -File `"$ScriptsDir\install_halfhour_push_test.ps1`""
Write-Host "  dir:"
Get-ChildItem $ScriptsDir -Filter "*halfhour*" | Format-Table Name,Length
